#!/bin/sh
# Type-check the frontend before committing (same check `npm run build` runs).
# Skip in an emergency with: git commit --no-verify

# Only run when the commit touches the frontend
git diff --cached --name-only --diff-filter=ACMRD | grep -q '^frontend/' || exit 0

ROOT=$(git rev-parse --show-toplevel)
echo "pre-commit: type-checking frontend..."

if [ -x "$ROOT/frontend/node_modules/.bin/tsc" ]; then
	(cd "$ROOT/frontend" && ./node_modules/.bin/tsc --noEmit -p .)
elif docker compose -f "$ROOT/compose.yaml" ps --status running frontend 2>/dev/null | grep -q frontend; then
	docker compose -f "$ROOT/compose.yaml" exec -T frontend npx tsc --noEmit -p .
else
	echo "pre-commit: no tsc found (run 'npm install' in frontend/ or 'make up'). Skipping."
	exit 0
fi

status=$?
if [ $status -ne 0 ]; then
	echo ""
	echo "pre-commit: TypeScript errors found, commit aborted."
	echo "            Fix them or use 'git commit --no-verify' to skip."
fi
exit $status
